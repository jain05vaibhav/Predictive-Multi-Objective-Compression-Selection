1. ENVIRONMENT SETUP & DEPENDENCIES
--------------------------------------------------------------------------------
# Create Python Virtual Environment:
python -m venv .venv

# Activate Virtual Environment (Windows PowerShell):
.\.venv\Scripts\Activate.ps1

# Activate Virtual Environment (Windows CMD):
.\.venv\Scripts\activate.bat

# Activate Virtual Environment (Linux / Raspberry Pi Bash):
source .venv/bin/activate

# Install Project Requirements:
pip install -r requirements.txt

# (Raspberry Pi 3B+ Only) Hardware Libraries & Interfaces:
sudo raspi-config                             # Enable Camera interface
sudo apt-get update
sudo apt-get install -y python3-picamera2 libgpiod2 libraspberrypi-bin
pip install adafruit-circuitpython-dht opencv-python psutil


2. SENSOR & TELEMETRY MODULE EXECUTION
--------------------------------------------------------------------------------
# Test Raspberry Pi 3B+ Native System Telemetry Reader (vcgencmd, psutil, /sys, /proc):
python -m edge.sensors.rpi_system_reader

# Test Unified Hardware Telemetry Hub (DHT22 + RPi 3B+ System + Camera):
python -m edge.sensors.telemetry_source

# Test CSI / USB Camera Reader (Captures in RAM & overwrites latest_frame.jpg):
python -m edge.sensors.camera_reader

# Test DHT22 Environmental Sensor (GPIO4):
python -m edge.sensors.dht22_reader


3. PIPELINE STAGE EXECUTION
--------------------------------------------------------------------------------
# Run Stage 1 Data Acquisition & Windowing Standalone:
python -m edge.stage1_acquisition

# Run Stage 2 Feature Extraction Standalone (Entropy, Variance, RoC):
python -m edge.stage2_features

# Run Stage 3 State & Resource Predictor Standalone (EWMA & Trend Forecasting):
python -m edge.stage3_predictor

# Run Stage 1 + Stage 2 + Stage 3 End-to-End Inline Test:
python -c "from edge.stage1_acquisition import AcquisitionStage; from edge.stage2_features import FeatureExtractionStage; from edge.stage3_predictor import PredictorStage; s1 = AcquisitionStage(window_size=10); s2 = FeatureExtractionStage(); s3 = PredictorStage(); win = s1.acquire_window(); feats = s2.extract_features(win); pred = s3.predict(win); print('Acquired Window:', win); print('Features:', feats); print('Predicted Next State:', pred)"

--------------------------------------------------------------------------------

python -m unittest discover tests

python -m unittest discover tests -v

python -m unittest tests/test_rpi_system_reader.py

# Run Stage 1 Tests Only:
python -m unittest tests/test_stage1.py

# Run Stage 2 Tests Only:
python -m unittest tests/test_stage2.py

python -m unittest tests/test_stage3.py

python -m unittest tests/test_camera_reader.py
